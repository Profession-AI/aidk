204    def _initialize_or_create_chat(self, chat_id: str, system_prompt: SystemPrompt) -> str:
205        """Initialize existing chat or create new one with specified ID.
206        
207        Parameters
208        ----------
209        chat_id : str
210            The chat ID to check/create
211        system_prompt : SystemPrompt
212            System prompt to use if creating new chat
213            
214        Returns
215        -------
216        str
217            The chat ID (either existing or newly created)
218        """
219        try:
220            # Try to load existing chat
221            existing_messages = self._history.get_messages(chat_id)
222            
223            # Check if chat exists and has messages
224            if existing_messages and len(existing_messages) > 0:
225                logger.info(f"Loaded existing chat: {chat_id}")
226                return chat_id
227            else:
228                # Chat doesn't exist, create it with the specified ID
229                logger.info(f"Creating new chat with ID: {chat_id}")
230                return self._create_chat_with_id(chat_id, system_prompt)
231                
232        except Exception as e:
233            # If loading fails, assume chat doesn't exist and create it
234            logger.info(f"Failed to load chat {chat_id}, creating new one: {e}")
235            return self._create_chat_with_id(chat_id, system_prompt)

380    def _process_file_attachment(self, prompt: str, file: Optional[Union[str, bytes]], file_type: Optional[str]) -> Union[str, List[Dict[str, Any]]]:
381        """Process file attachment and return modified prompt.
382        
383        Parameters
384        ----------
385        prompt : str
386            Original prompt text
387        file : Optional[Union[str, bytes]]
388            File to attach (path, URL, or bytes)
389        file_type : Optional[str]
390            Type of the file
391            
392        Returns
393        -------
394        Union[str, List[Dict[str, Any]]]
395            Modified prompt with file content or multimodal content
396            
397        Raises
398        ------
399        ChatError
400            If file processing fails
401        """
402        if file is None:
403            return prompt
404            
405        try:
406            # Determine file type
407            if file_type is None and isinstance(file, str):
408                file_type = file.split(".")[-1].lower()
409                
410            if not file_type:
411                raise ChatError("File type could not be determined")
412                
413            conf = Conf()
414            
415            # Handle text files
416            if file_type in conf["supported_files"]["text"]:
417                return self._process_text_file(prompt, file)
418            # Handle image files
419            elif file_type in conf["supported_files"]["image"]:
420                return self._process_image_file(prompt, file)
421            else:
422                raise ChatError(f"Unsupported file type: {file_type}")
423                
424        except Exception as e:
425            raise ChatError(f"Failed to process file attachment: {e}")

427    def _process_text_file(self, prompt: str, file: Union[str, bytes]) -> str:
428        """Process text file attachment.
429        
430        Parameters
431        ----------
432        prompt : str
433            Original prompt text
434        file : Union[str, bytes]
435            Text file to process
436            
437        Returns
438        -------
439        str
440            Prompt with file content appended
441            
442        Raises
443        ------
444        ChatError
445            If text file processing fails
446        """
447        try:
448            if isinstance(file, str):
449                # Handle as file path or URL
450                if os.path.exists(file):
451                    with open(file, "r", encoding="utf-8") as f:
452                        file_content = f.read()
453                else:
454                    # Assume it's a URL or remote file
455                    file_content = file
456            else:
457                # Handle as bytes
458                file_content = file.decode("utf-8")
459                
460            return prompt + Conf()["default_prompt"]["file"] + file_content
461            
462        except Exception as e:
463            raise ChatError(f"Failed to process text file: {e}")

465    def _process_image_file(self, prompt: str, file: Union[str, bytes]) -> List[Dict[str, Any]]:
466        """Process image file attachment.
467        
468        Parameters
469        ----------
470        prompt : str
471            Original prompt text
472        file : Union[str, bytes]
473            Image file to process
474            
475        Returns
476        -------
477        List[Dict[str, Any]]
478            Multimodal content with text and image
479            
480        Raises
481        ------
482        ChatError
483            If image file processing fails
484        """
485        try:
486            if isinstance(file, str):
487                # Handle as file path or URL
488                if os.path.exists(file):
489                    # Convert local file to base64
490                    with open(file, "rb") as f:
491                        file_bytes = f.read()
492                    image_url = f"data:image/jpeg;base64,{base64.b64encode(file_bytes).decode('utf-8')}"
493                else:
494                    # Assume it's already a URL
495                    image_url = file
496            else:
497                # Handle as bytes
498                image_url = f"data:image/jpeg;base64,{base64.b64encode(file).decode('utf-8')}"
499                
500            return [
501                {"type": "text", "text": prompt},
502                {"type": "image_url", "image_url": image_url}
503            ]
504            
505        except Exception as e:
506            raise ChatError(f"Failed to process image file: {e}")

509        """Prepare messages for the API call.

573    def send(self, prompt: str, file: Optional[Union[str, bytes]] = None, file_type: Optional[str] = None, return_history: bool = False, metadata: Optional[Dict[str, Any]] = {}) -> ChatResponse:
574        """
575        Send the model a question.
576
577        Parameters
578        ----------
579        prompt : str
580            The question to send the model
581        file : str | bytes, optional
582            The file to attach to the message, if it's a string it will be treated as a local path or remote url to a file
583        file_type : str, optional
584            The type of the file
585        return_history : bool, optional
586            Whether to return the full history of the chat or only the response
587        metadata : Dict[str, Any], optional
588            Metadata to pass to the model for observability and tracking
589
590        Returns
591        -------
592        ChatResponse
593            ChatResponse dataclass containing:
594            - chat_id: str - The chat identifier
595            - response: ModelResponse - The model response
596            - history: Optional[List[Message]] - Full chat history if return_history is True
597
598        Raises
599        ------
600        ChatError
601            If the request fails or parameters are invalid
602        """
603
604        metadata = self._metadata | metadata
605
606        try:
607            # Process file attachment
608            processed_prompt = self._process_file_attachment(prompt, file, file_type)
609            
610            # Prepare messages
611            messages = self._prepare_messages(processed_prompt)
612
613            # Make API call
614            model_response = self._model.ask(messages, metadata=metadata)
615            
616            # Store messages
617            self._store_messages(messages, model_response.response)
618            
619            # Get history if requested
620            history = None
621            if return_history:
622                history = self._history.get_messages(self.chat_id)
623            
624            return ChatResponse(
625                chat_id=self.chat_id,
626                history=history,
627                **asdict(model_response)
628            )
629                        
630        except Exception as e:
631            logger.error(f"Ask request failed: {e}")
632            raise ChatError(f"Ask request failed: {e}")

692    async def send_async(self, prompt: str, file: Optional[Union[str, bytes]] = None, file_type: Optional[str] = None, metadata: Optional[Dict[str, Any]] = None) -> "ChatResponse":
693        """
694        Send the model a question using the async (non-streaming) model API and return a ChatResponse.
695
696        This method awaits the model's async completion and returns a `ChatResponse` wrapping
697        the underlying `ModelResponse`.
698        """
699
700        # Merge metadata safely
701        metadata = self._metadata | (metadata or {})
702
703        try:
704            # Process file attachment
705            processed_prompt = self._process_file_attachment(prompt, file, file_type)
706
707            # Prepare messages
708            messages = self._prepare_messages(processed_prompt)
709
710            # Await the model's async (non-streaming) API
711            model_response = await self._model.ask_async(messages, metadata=metadata)
712
713            # Store messages (use the string payload from ModelResponse)
714            self._store_messages(messages, model_response.response)
715
716            # Get history if requested (not applicable here, but kept for parity)
717            history = self._history.get_messages(self.chat_id) if hasattr(self._history, 'load') else None
718
719            return ChatResponse(
720                chat_id=self.chat_id,
721                history=history,
722                **asdict(model_response)
723            )
724
725        except Exception as e:
726            logger.error(f"Async request failed: {e}")
727            raise ChatError(f"Async request failed: {e}")

Chat class is responsible for handling the chat interface and messages history.

Basic usage:

from aidk.models import Model
from aidk.chat import Chat
model = Model(provider="openai", model="gpt-4o-mini")
chat = Chat(model=model)

response = chat.send("+2") # {"chat_id": "...", "response": "6", "model": {...}}
