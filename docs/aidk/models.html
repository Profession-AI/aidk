 18
 19    Examples
 20    --------
 21    Basic usage with ElevenLabs:
 22    >>> model = VoiceModel(
 23    ...     provider="elevenlabs",
 24    ...     model="eleven_multilingual_v2",
 25    ...     voice="21m00Tcm4TlvDq8ikWAM"
 26    ... )
 27    >>> audio_file = model.speak("Hello, world!")
 28
 29    Using different return types:
 30    >>> # Save as MP3 file
 31    >>> audio_file = model.speak("Hello, world!", return_type="output.mp3")
 32
 33    >>> # Get as base64 string
 34    >>> audio_b64 = model.speak("Hello, world!", return_type="base64")
 35
 36    >>> # Get as bytes
 37    >>> audio_bytes = model.speak("Hello, world!", return_type="bytes")
 38
 39    Streaming audio generation:
 40    >>> async for audio_chunk in model.stream("Long text to convert to speech"):
 41    ...     print(f"Generated audio chunk: {audio_chunk}")
 42    """

 44    def __init__(self, provider: str, model: str, voice: str):
 45        """
 46        Initialize a new VoiceModel instance.
 47
 48        Parameters
 49        ----------
 50        provider : str
 51            The voice provider to use (e.g., 'elevenlabs', 'openai')
 52        model : str
 53            The specific voice model to use
 54        voice : str
 55            The voice ID or voice name to use for speech generation
 56
 57        Raises
 58        ------
 59        ImportError
 60            If the required provider library is not installed
 61        """
 62        load_key(provider)
 63        self._provider = provider
 64        self._model = model
 65        self._voice = voice

 67        if self._provider == "elevenlabs":
 68            try:
 69                from elevenlabs.client import ElevenLabs

 78    def speak(self, text: str, min_chars_per_sentence=100, return_type="audio.mp3"):
 79        """
 80        Convert text to speech and return the audio in the specified format.
 81        
 82        The method automatically splits long text into optimal chunks based on
 83        sentence boundaries to ensure high-quality audio generation.
 84        
 85        Parameters
 86        ----------
 87        text : str
 88            The text to convert to speech
 89        min_chars_per_sentence : int, optional
 90            Minimum number of characters per sentence group for optimal audio
 91            generation (default: 100)
 92        return_type : str, optional
 93            The format to return the audio in. Options:
 94            - File path ending with '.mp3' or '.wav' to save to file
 95            - 'base64' to return as base64 encoded string
 96            - 'bytes' to return as raw bytes (default: "audio.mp3")
 97            
 98        Returns
 99        -------
100        str or bytes
101            The generated audio in the specified format:
102            - If return_type is a file path: returns the file path
103            - If return_type is 'base64': returns base64 encoded string
104            - If return_type is 'bytes': returns raw audio bytes
105            
106        Raises
107        ------
108        ValueError
109            If return_type is not a valid option
110            
111        Examples
112        --------
113        >>> model = VoiceModel(
114        ...     provider="elevenlabs",
115        ...     model="eleven_multilingual_v2",
116        ...     voice="21m00Tcm4TlvDq8ikWAM"
117        ... )
118        >>> # Save to file
119        >>> audio_file = model.speak("Hello, world!", return_type="output.mp3")
120        >>> # Get as base64
121        >>> audio_b64 = model.speak("Hello, world!", return_type="base64")
122        >>> # Get as bytes
123        >>> audio_bytes = model.speak("Hello, world!", return_type="bytes")
124        """
125        audio_groups = self._generate_audio_groups(text, min_chars_per_sentence)
126        audio_chunks = []
127
128        for group in audio_groups:
129            response = self._generate(group)
130            audio_chunks.append(response)
131
132        if return_type.endswith(".mp3") or return_type.endswith(".wav"):
133            combined_audio = self._combine_bytes_chunks(audio_chunks)
134            with open(return_type, 'wb') as f:
135                f.write(combined_audio)
136            return return_type
137        if return_type == "base64":
138            combined_audio = self._combine_bytes_chunks(audio_chunks)
139            return base64.b64encode(combined_audio).decode('utf-8')
140        if return_type == "bytes":
141            return self._combine_bytes_chunks(audio_chunks)
142        raise ValueError(f"Invalid return type: {return_type}")
143
144    def _combine_bytes_chunks(self, audio_chunks: list) -> bytes:
145        """
146        Combine audio chunks into a single bytes object.
147
148        Parameters
149        ----------
150        audio_chunks : list
151            List of audio byte chunks to combine
152
153        Returns
154        -------
155        bytes
156            Combined audio data as bytes
157        """
158        combined = b""
159        for chunk in audio_chunks:
160            combined += chunk
161        return combined

163    def _generate_audio_groups(self, text: str, min_chars_per_sentence: int):
164        """
165        Generate optimized sentence groups for audio generation.
166        
167        Splits text into sentence groups that meet the minimum character requirement
168        for optimal audio quality while respecting sentence boundaries.
169        
170        Parameters
171        ----------
172        text : str
173            The text to split into groups
174        min_chars_per_sentence : int
175            Minimum number of characters per sentence group
176            
177        Returns
178        -------
179        list
180            List of sentence groups optimized for audio generation
181        """
182        sentences = [s.strip() for s in text.split(".") if s.strip()]
183        audio_groups = []
184        i = 0
185
186        while i < len(sentences):
187            current_group = sentences[i]
188            j = i + 1
189
190            # Continue adding subsequent sentences until min_chars_per_sentence is exceeded
191            while j < len(sentences):
192                next_sentence = sentences[j]
193                combined_length = len(current_group + ". " + next_sentence)
194
195                # If the combination exceeds the limit, stop
196                if combined_length > min_chars_per_sentence:
197                    break
198
199                # Otherwise, join the next sentence
200                current_group += ". " + next_sentence
201                j += 1
202
203            audio_groups.append(current_group)
204            # Move to the next unprocessed sentence
205            i = j
206
207        return audio_groups

215        text : str

217            
218        Returns
219        -------

222        """

225                text=text,

228                output_format="mp3_44100_128",
229            )
230            response_bytes = b""
231            for r in response:
232                response_bytes += r
233            return response_bytes

235            model=self._provider + "/" + self._model,

A class for text-to-speech using various AI voice models.

VoiceModel provides an interface for converting text to speech using different AI voice providers. Currently supports ElevenLabs and OpenAI voice models.

Examples

Basic usage with ElevenLabs:

>>> model = VoiceModel(
...     provider="elevenlabs",
...     model="eleven_multilingual_v2",
...     voice="21m00Tcm4TlvDq8ikWAM"

Using different return types:

>>> # Save as MP3 file
>>> audio_file = model.speak("Hello, world!", return_type="output.mp3")

>>> # Get as base64 string

>>> # Get as bytes
>>> audio_bytes = model.speak("Hello, world!", return_type="bytes")
