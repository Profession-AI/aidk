102        """
103        Perform a semantic search query against the vector database.
104        
105        This method converts the input query into a vector embedding and searches
106        the database for the most semantically similar documents.
107        
108        Parameters:
109        -----------
110        query : str
111            The text query to search for. This will be converted to a vector
112            embedding and used to find similar documents.
113            
114        top_k : int, default=10
115            The number of most relevant documents to return. Higher values
116            return more results but may include less relevant documents.
117            
118        Returns:
119        --------
120        List[DocumentRetrieved]
121            A list of DocumentRetrieved objects, each containing:
122            - content: The document text
123            - metadata: Dictionary of document metadata
124            - doc_id: Unique document identifier
125            - distance: Similarity score (lower = more similar, 0 = identical)
126            
127        Examples:
128        ---------
129        ```python
130        # Basic query
131        results = rag.query("What is artificial intelligence?")
132        for doc in results:
133            print(f"ID: {doc.doc_id}")
134            print(f"Content: {doc.content}")
135            print(f"Similarity: {doc.distance}")
136            print(f"Metadata: {doc.metadata}")
137        
138        # Query with more results
139        results = rag.query("Machine learning algorithms", top_k=20)
140        ```
141        
142        Notes:
143        ------
144        - Results are returned in order of relevance (most similar first)
145        - Distance scores are cosine distances (0 = identical, 2 = completely opposite)
146        - If fewer than top_k documents exist in the database, all available documents are returned
147        """
148        return self._vector_db.query(query, top_k)

Retrieval-Augmented Generation (RAG) system for semantic search and document retrieval.

This class provides a high-level interface for performing semantic search queries against a vector database. It supports multiple vector database backends and embedding providers for flexible deployment scenarios.

The RAG system works by:

  1. Converting text queries into vector embeddings
  2. Searching the vector database for similar document embeddings
  3. Returning the most relevant documents based on semantic similarity

Attributes: _vectorizer (str): The embedding model used for vectorization _db (str): Name of the vector database _vector_db (ChromaVectorDB): The vector database backend

 34    # Add documents

 36    metadatas = [{"source": "file1.txt"}, {"source": "file2.txt"}]
 37    ids = ["doc1", "doc2"]
 38    
 39    vector_db.add(documents, metadatas, ids)

 41    # Search for similar documents
 42    results = vector_db.query("search query", k=5)

 46    
 47    ```python
 48    # Initialize with OpenAI embeddings
 49    vector_db = ChromaVectorDB(

 60        """
 61        Initialize the ChromaDB vector database.
 62        
 63        Parameters:
 64        -----------
 65        name : str, optional
 66            Name of the ChromaDB collection. If provided, the collection

104    def add(self, documents: List[Document]) -> None:
105        """
106        Add documents to the ChromaDB collection.
107        
108        This method adds Document objects to the ChromaDB collection.
109        The documents are automatically converted to embeddings using the
110        configured embedding model.
111        
112        Parameters:
113        -----------
114        documents : List[Document]
115            List of Document objects to add to the database.
116            Each Document contains content, metadata, and a unique ID.
117            
118        Examples:
119        ---------
120        ```python
121        from aidk.rag.documents_builder import DocumentsBuilder, WordChunker, Document
122        
123        # Using Document objects from DocumentsBuilder
124        chunker = WordChunker(chunk_size=1000, chunk_overlap=100)
125        builder = DocumentsBuilder(chunker=chunker)
126        docs = builder.from_file("document.txt")
127        
128        # Add to vector database
129        vector_db.add(docs)
130        
131        # Or create Document objects manually
132        custom_docs = [
133            Document(
134                content="Machine learning content",
135                metadata={"topic": "ml", "source": "textbook"},
136                doc_id="doc_001"
137            )
138        ]
139        vector_db.add(custom_docs)
140        ```
141        """
142        doc_contents = [doc.content for doc in documents]
143        doc_metadatas = [doc.metadata for doc in documents]
144        doc_ids = [doc.doc_id for doc in documents]
145        
146        if not (len(doc_contents) == len(doc_metadatas) == len(doc_ids)):
147            raise ValueError("All documents must have consistent structure")
148            
149        self._collection.add(
150            documents=doc_contents,
151            metadatas=doc_metadatas,
152            ids=doc_ids
153        )

155    def query(self, query: str, top_k: int = 10) -> List[DocumentRetrieved]:
156        """
157        Search for similar documents in the ChromaDB collection.
158        
159        This method performs semantic search by converting the query to an
160        embedding and finding the most similar document embeddings in the
161        collection.
162        
163        Parameters:
164        -----------
165        query : str
166            The text query to search for. This will be converted to a
167            vector embedding and compared against stored documents.
168            
169        top_k : int, default=10
170            Number of most similar documents to return. Higher values
171            return more results but may include less relevant documents.
172            
173        Returns:
174        --------
175        List[DocumentRetrieved]
176            List of DocumentRetrieved objects containing the retrieved documents,
177            their metadata, IDs, and similarity distances.
178            
179        Examples:
180        ---------
181        ```python
182        # Basic search
183        results = vector_db.query("What is machine learning?", top_k=5)
184        
185        # Access results
186        for doc in results:
187            print(f"Content: {doc.content}")
188            print(f"Distance: {doc.distance}")
189            print(f"ID: {doc.doc_id}")
190            print(f"Metadata: {doc.metadata}")
191        ```
192        
193        Notes:
194        ------
195        - Results are returned in order of similarity (most similar first)
196        - Distance scores are cosine distances (0 = identical, 2 = opposite)
197        - If fewer than top_k documents exist, all available documents are returned
198        - The query is automatically embedded using the same model as stored documents
199        """
200        results = self._collection.query(
201            query_texts=query,
202            n_results=top_k
203        )
204        
205        retrieved_docs = []
206        if results and results['ids'] and len(results['ids']) > 0:
207            for i, doc_id in enumerate(results['ids'][0]):
208                retrieved_docs.append(
209                    DocumentRetrieved(
210                        content=results['documents'][0][i],
211                        metadata=results['metadatas'][0][i],
212                        doc_id=doc_id,
213                        distance=results['distances'][0][i]
214                    )
215                )
216        
217        return retrieved_docs

211                        metadata=results['metadatas'][0][i],
212                        doc_id=doc_id,
213                        distance=results['distances'][0][i]
214                    )
215                )
216
